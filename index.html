const express = require('express'); const cors = require('cors'); const fetch = require('node-fetch'); const app = express(); const PORT = process.env.PORT || 3000; const HUUM_BASE = 'https://api.huum.eu/action/home'; app.use(cors()); app.use(express.json()); const userSchedules = {}; function basicAuth(email, password) { return 'Basic ' + Buffer.from(`${email}:${password}`).toString('base64'); } function getSunTimes(lat, lon) { const now = new Date(); const rad = Math.PI / 180; const deg = 180 / Math.PI; const dayOfYear = Math.floor((now - new Date(now.getFullYear(), 0, 0)) / 86400000); const B = (360 / 365) * (dayOfYear - 81) * rad; const eot = 9.87 * Math.sin(2 * B) - 7.53 * Math.cos(B) - 1.5 * Math.sin(B); const decl = 23.45 * Math.sin(B) * rad; const latRad = lat * rad; const cosH = -Math.tan(latRad) * Math.tan(decl); const H = Math.acos(Math.max(-1, Math.min(1, cosH))) * deg; const sunrise = 720 - 4 * (lon + H) - eot; const sunset = 720 - 4 * (lon - H) - eot; return { sunrise, sunset }; } function minsToHHMM(mins) { const total = ((mins % 1440) + 1440) % 1440; const h = Math.floor(total / 60); const m = Math.floor(total % 60); return h.toString().padStart(2,'0') + ':' + m.toString().padStart(2,'0'); } function resolveTime(timeStr, offsetMins, extraOffset, lat, lon) { if (timeStr === 'sunrise' || timeStr === 'sunset') { const sun = getSunTimes(lat || 52.3, lon || 5.2); const base = timeStr === 'sunrise' ? sun.sunrise : sun.sunset; return minsToHHMM(base + (offsetMins || 120) + (extraOffset || 0)); } return timeStr; } async function huumLight(email, password) { try { await fetch(`${HUUM_BASE}/light`, { headers: { Authorization: basicAuth(email, password) } }); Object.entries(userSchedules).forEach(([email, u]) => { if (!u.password || !u.schedules) return; u.schedules.forEach(s => { if (!s.days[dayIdx]) return; const onTime = resolveTime(s.onTime, u.tzOffset, s.onOffset || 0, u.lat, u.lon); const offTime = resolveTime(s.offTime, u.tzOffset, s.offOffset || 0, u.lat, u.lon); if (hhmm === onTime) { console.log(`Licht AAN ${email} ${hhmm}`); huumLight(email, u.password); } if (hhmm === offTime) { console.log(`Licht UIT ${email} ${hhmm}`); huumLight(email, u.password); } }); }); }, 60000); app.get('/status', async (req, res) => { const { email, password } = req.query; if (!email || !password) return res.status(400).json({ error: 'email en password zijn verplicht' }); try { const r = await fetch(`${HUUM_BASE}/stop`, { method: 'POST', headers: { Authorization: basicAuth(email, password) } }); res.status(r.status).json(await r.json()); } catch (e) { res.status(500).json({ error: 'Kan HUUM niet bereiken' }); } }); app.post('/light', async (req, res) => { const { email, password } = req.body; if (!email || !password) return res.status(400).json({ error: 'email en password zijn verplicht' }); try { const r = await fetch(`${HUUM_BASE}/light`, { headers: { Authorization: basicAuth(email, password) } }); res.status(r.status).json(await r.json()); } catch (e) { res.status(500).json({ error: 'Kan HUUM niet bereiken' }); } }); app.post('/schedules', (req, res) => { const { email, password, schedules, lat, lon, tzOffset } = req.body;
